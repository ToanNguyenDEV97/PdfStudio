    private MainViewModel? GetMainVm() => Window.GetWindow(this)?.DataContext as MainViewModel;

    private bool IsHandModeActive()
    {
        var mainVm = GetMainVm();
        return mainVm?.ActiveTool == ViewerToolMode.Hand || Keyboard.IsKeyDown(Key.Space);
    }

    private void OnDataContextChanged(object sender, DependencyPropertyChangedEventArgs e)
    {
        if (_currentTab is not null)
            _currentTab.ScrollToPageRequested -= OnScrollToPageRequested;

        _currentTab = e.NewValue as DocumentTabViewModel;

        if (_currentTab is not null)
        {
            _currentTab.UpdateDpiScale(VisualTreeHelper.GetDpi(this).DpiScaleX);
            _currentTab.ScrollToPageRequested += OnScrollToPageRequested;
            MainScrollViewer.ScrollToVerticalOffset(0);
            Dispatcher.InvokeAsync(TriggerVisibleRender);
        }
    }

    private void OnScrollToPageRequested(int pageIndex)
    {
        if (_currentTab is null || pageIndex < 0 || pageIndex >= _currentTab.Pages.Count)
            return;

        double yOffset = TopPadding;
        for (int i = 0; i < pageIndex; i++)
            yOffset += _currentTab.Pages[i].DisplayHeight + PageSpacing;

        MainScrollViewer.ScrollToVerticalOffset(yOffset);
    }

    private void MainScrollViewer_ScrollChanged(object sender, ScrollChangedEventArgs e) =>
        TriggerVisibleRender();

    private void MainScrollViewer_PreviewMouseWheel(object sender, MouseWheelEventArgs e)
    {
        if (_currentTab is null) return;

        if ((Keyboard.Modifiers & ModifierKeys.Control) == ModifierKeys.Control)
        {
            e.Handled = true;
            double step = e.Delta > 0 ? 15 : -15;
            _currentTab.ZoomPercent = Math.Clamp(_currentTab.ZoomPercent + step, 25, 400);
        }
    }

        MainScrollViewer.ScrollToHorizontalOffset(_panStartHOffset - (current.X - _panStartPoint.X));

    }

 private async void PageOverlay_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
    {
        var mainVm = GetMainVm();
        if (_currentTab is null || mainVm is null || IsHandModeActive()) return;
        if (sender is not Canvas canvas || canvas.Tag is not int pageIndex) return;

        Point pt = e.GetPosition(canvas);
        _currentTab.ClearSelection();
        _currentTab.SelectAnnotation(null);

        switch (mainVm.ActiveTool)
        {
            case ViewerToolMode.SelectText:
            case ViewerToolMode.Highlight:
            case ViewerToolMode.Underline:
            {
                var chars = await _currentTab.GetPageCharsAsync(pageIndex);
                if (chars.Count == 0) return;

                int hitIndex = FindClosestCharIndex(chars, pt);
                if (hitIndex < 0) return;

                _isSelectingText = true;
                _selectingPageIndex = pageIndex;
                _selectionAnchorCharIndex = hitIndex;
                _activePageChars = chars;
                canvas.CaptureMouse();
                break;
            }
            case ViewerToolMode.StickyNote:
            {
                var prompt = new PromptDialog("ThÃªm Sticky Note", "Nháº­p ná»™i dung ghi chÃº:")
                {
                    Owner = Window.GetWindow(this)
                };
                if (prompt.ShowDialog() == true && !string.IsNullOrWhiteSpace(prompt.InputValue))
                {
                    _currentTab.AddAnnotation(new PdfAnnotationModel
                    {
                        PageIndex = pageIndex,
                        Type = AnnotationType.StickyNote,
                        Bounds = new PdfRect(pt.X - 11, pt.Y - 11, 22, 22),
                        TextContent = prompt.InputValue,
                        ColorHex = _currentTab.SelectedColorHex
                    });
                }
                break;
            }
            case ViewerToolMode.FreeText:
            {
                var prompt = new PromptDialog("ThÃªm chá»¯ (FreeText)", "Nháº­p ná»™i dung vÄƒn báº£n muá»‘n chÃ¨n:")
                {
                    Owner = Window.GetWindow(this)
                };
                if (prompt.ShowDialog() == true && !string.IsNullOrWhiteSpace(prompt.InputValue))
                {
                    double fontSize = DocumentTabViewModel.ParseFontSize(_currentTab.SelectedFontSize, 13);
                    double estWidth = Math.Clamp(prompt.InputValue.Length * fontSize * 0.62 + 16, 90, 420);
                    double estHeight = fontSize * 1.7 + 8;

                    _currentTab.AddAnnotation(new PdfAnnotationModel
                    {
                        PageIndex = pageIndex,
                        Type = AnnotationType.FreeText,
                        Bounds = new PdfRect(pt.X, pt.Y - fontSize * 0.5, estWidth, estHeight),
                        TextContent = prompt.InputValue,
                        ColorHex = _currentTab.SelectedColorHex,
                        FontName = _currentTab.SelectedFont,
                        FontSize = fontSize
                    });
                }
                break;
            }
            case ViewerToolMode.Ink:
            {
                _isDrawingInk = true;
                _inkPageIndex = pageIndex;
                _activeInkPoints.Clear();
                _activeInkPoints.Add(new PdfPoint(pt.X, pt.Y));

                _activeInkPolyline = canvas.Children.OfType<Polyline>().LastOrDefault();
                if (_activeInkPolyline is not null)
                {
                    try
                    {
                        _activeInkPolyline.Stroke = new SolidColorBrush((Color)ColorConverter.ConvertFromString(_currentTab.SelectedColorHex));
                    }
                    catch { _activeInkPolyline.Stroke = Brushes.Crimson; }

                    _activeInkPolyline.StrokeThickness = _currentTab.SelectedStrokeThickness;
                    _activeInkPolyline.Points = [pt];
                }
                canvas.CaptureMouse();
                break;
            }
        }
    }

    private void PageOverlay_MouseMove(object sender, MouseEventArgs e)
    {
        if (sender is not Canvas canvas) return;
        var mainVm = GetMainVm();

        if (_isDrawingInk && _activeInkPolyline is not null)
        {
            Point pt = e.GetPosition(canvas);
            var last = _activeInkPoints[^1];
            if ((pt.X - last.X) * (pt.X - last.X) + (pt.Y - last.Y) * (pt.Y - last.Y) >= 2.25)
            {
                _activeInkPoints.Add(new PdfPoint(pt.X, pt.Y));
                _activeInkPolyline.Points.Add(pt);
            }
            return;
        }

        if (!_isSelectingText)
        {
            canvas.Cursor = IsHandModeActive() ? Cursors.Hand
                : mainVm?.ActiveTool is ViewerToolMode.SelectText or ViewerToolMode.Highlight or ViewerToolMode.Underline ? Cursors.IBeam
                : mainVm?.ActiveTool is ViewerToolMode.Ink or ViewerToolMode.StickyNote or ViewerToolMode.FreeText ? Cursors.Cross
                : Cursors.Arrow;
            return;
        }

        if (_currentTab is null || _activePageChars is null || _activePageChars.Count == 0) return;

        Point currentPt = e.GetPosition(canvas);
        int currentIndex = FindClosestCharIndex(_activePageChars, currentPt);
        if (currentIndex < 0) return;

        int from = Math.Min(_selectionAnchorCharIndex, currentIndex);
        int to = Math.Max(_selectionAnchorCharIndex, currentIndex);

        var rects = DocumentTabViewModel.MergeCharRectsByLine(_activePageChars, from, to);
        var sb = new StringBuilder(to - from + 1);
        for (int i = from; i <= to; i++)
            sb.Append(_activePageChars[i].Character);

        _currentTab.ApplySelection(_selectingPageIndex, rects, sb.ToString());
    }

    private void PageOverlay_MouseLeftButtonUp(object sender, MouseButtonEventArgs e)
    {
        var mainVm = GetMainVm();

        if (_isDrawingInk)
        {
            _isDrawingInk = false;
            if (sender is Canvas inkCanvas) inkCanvas.ReleaseMouseCapture();
            _activeInkPolyline?.Points.Clear();

            if (_currentTab is not null && _activeInkPoints.Count >= 2)
            {
                double pad = Math.Max(4, _currentTab.SelectedStrokeThickness);
                double minX = _activeInkPoints.Min(p => p.X) - pad;
                double minY = _activeInkPoints.Min(p => p.Y) - pad;
                double maxX = _activeInkPoints.Max(p => p.X) + pad;
                double maxY = _activeInkPoints.Max(p => p.Y) + pad;

                _currentTab.AddAnnotation(new PdfAnnotationModel
                {
                    PageIndex = _inkPageIndex,
                    Type = AnnotationType.Ink,
                    Bounds = new PdfRect(minX, minY, maxX - minX, maxY - minY),
                    InkPoints = [.. _activeInkPoints],
                    ColorHex = _currentTab.SelectedColorHex,
                    StrokeThickness = _currentTab.SelectedStrokeThickness
                });
            }
            _activeInkPoints.Clear();
            return;
        }

        if (!_isSelectingText) return;
        _isSelectingText = false;
        if (sender is Canvas canvas) canvas.ReleaseMouseCapture();

        // Náº¿u Ä‘ang báº­t cÃ´ng cá»¥ Highlight hoáº·c Gáº¡ch dÆ°á»›i -> chuyá»ƒn vÃ¹ng vá»«a bÃ´i Ä‘en thÃ nh Annotation
        if (_currentTab is not null && mainVm?.ActiveTool is ViewerToolMode.Highlight or ViewerToolMode.Underline)
        {
            var pageVm = _currentTab.Pages[_selectingPageIndex];
            if (pageVm.SelectionRects.Count > 0)
            {
                var quads = pageVm.SelectionRects
                    .Select(r => new PdfRect(r.X, r.Y, r.Width, r.Height))
                    .ToList();

                double minX = quads.Min(r => r.X);
                double minY = quads.Min(r => r.Y);
                double maxX = quads.Max(r => r.X + r.Width);
                double maxY = quads.Max(r => r.Y + r.Height);

                _currentTab.AddAnnotation(new PdfAnnotationModel
                {
                    PageIndex = _selectingPageIndex,
                    Type = mainVm.ActiveTool == ViewerToolMode.Underline ? AnnotationType.Underline : AnnotationType.Highlight,
                    Bounds = new PdfRect(minX, minY, maxX - minX, maxY - minY),
                    QuadRects = quads,
                    ColorHex = _currentTab.SelectedColorHex,
                    StrokeThickness = _currentTab.SelectedStrokeThickness
                });
                _currentTab.ClearSelection();
            }
        }
    }

                annot.TextContent)
            {
                Owner = Window.GetWindow(this)
            };
            if (prompt.ShowDialog() == true && !string.IsNullOrWhiteSpace(prompt.InputValue))
                _currentTab.UpdateAnnotationText(annot, prompt.InputValue);
            return;
        }

        // Cho phÃ©p kÃ©o di chuyá»ƒn Ä‘á»‘i tÆ°á»£ng
        _draggingPageCanvas = FindAncestorPageCanvas(el);

        _dragStartMouse = e.GetPosition(_draggingPageCanvas);

        Point current = e.GetPosition(_draggingPageCanvas);
        double newX = Math.Max(0, _dragOrigX + (current.X - _dragStartMouse.X));

    }

    private static int FindClosestCharIndex(IReadOnlyList<PageCharInfo> chars, Point pt)
    {
        int bestIndex = -1;
        double bestDistSq = double.MaxValue;

        for (int i = 0; i < chars.Count; i++)
        {
            var b = chars[i].Bounds;
            if (b.Width <= 0 || chars[i].Character == '\n') continue;

            // Khá»›p chÃ­nh xÃ¡c bÃªn trong khung kÃ½ tá»±
            if (pt.X >= b.X && pt.X <= b.X + b.Width && pt.Y >= b.Y && pt.Y <= b.Y + b.Height)
                return i;

            double cx = b.X + b.Width / 2.0;
            double cy = b.Y + b.Height / 2.0;
            double dx = pt.X - cx;
            double dy = (pt.Y - cy) * 2.5; // Æ¯u tiÃªn kÃ½ tá»± náº±m cÃ¹ng hÃ ng ngang
            double distSq = dx * dx + dy * dy;

            if (distSq < bestDistSq)
            {
                bestDistSq = distSq;
                bestIndex = i;
            }
        }

        return bestDistSq <= 45 * 45 ? bestIndex : -1;
    }

    private void TriggerVisibleRender()
    {
        if (_currentTab is null || _currentTab.Pages.Count == 0) return;

        double viewTop = MainScrollViewer.VerticalOffset;
        double viewBottom = viewTop + Math.Max(MainScrollViewer.ViewportHeight, 700);
        double viewCenter = (viewTop + viewBottom) / 2.0;

        int firstVisible = -1;
        int lastVisible = -1;
        int centerPage = 0;
        double currentY = TopPadding;

        for (int i = 0; i < _currentTab.Pages.Count; i++)
        {
            double pageHeight = _currentTab.Pages[i].DisplayHeight;
            double pageTop = currentY;
            double pageBottom = pageTop + pageHeight;

            if (pageBottom >= viewTop && pageTop <= viewBottom)
            {
                if (firstVisible == -1) firstVisible = i;
                lastVisible = i;
            }

            if (pageTop <= viewCenter && pageBottom >= viewCenter)
                centerPage = i;

            currentY = pageBottom + PageSpacing;
        }

        if (firstVisible == -1)
        {
            firstVisible = 0;
            lastVisible = 0;
        }

        if (_currentTab.CurrentPage != centerPage + 1)
            _currentTab.CurrentPage = centerPage + 1;

        _ = _currentTab.RequestVisiblePagesAsync(firstVisible, lastVisible);
    }

using System.Windows;
using System.Windows.Controls;
using System.Windows.Documents;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using PdfStudio.Core.Interfaces;
using PdfStudio.Core.Models;

namespace PdfStudio.App.Services;

public sealed class StreamingPrintPaginator : DocumentPaginator
{
    private const double PrintDpi = 200.0;
    private readonly IPdfReaderEngine _readerEngine;
    private readonly int _fromPageIndex;
    private readonly int _pageCount;
    private Size _pageSize;

    public StreamingPrintPaginator(
        IPdfReaderEngine readerEngine,
        Size printableAreaSize,
        int fromPageIndex,
        int toPageIndex)
    {
        _readerEngine = readerEngine;
        _pageSize = printableAreaSize;
        _fromPageIndex = Math.Clamp(fromPageIndex, 0, Math.Max(0, readerEngine.PageCount - 1));
        int safeTo = Math.Clamp(toPageIndex, _fromPageIndex, Math.Max(0, readerEngine.PageCount - 1));
        _pageCount = readerEngine.PageCount == 0 ? 0 : (safeTo - _fromPageIndex + 1);
    }

    public override bool IsPageCountValid => true;
    public override int PageCount => _pageCount;
    public override IDocumentPaginatorSource? Source => null;

    public override Size PageSize
    {
        get => _pageSize;
        set => _pageSize = value;
    }

    public override DocumentPage GetPage(int pageNumber)
    {
        int actualPageIndex = _fromPageIndex + pageNumber;
        double renderScale = PrintDpi / 72.0;

        RenderedPageResult raw = _readerEngine.RenderPage(actualPageIndex, renderScale);
        var bitmap = BitmapSource.Create(
            raw.Width, raw.Height, PrintDpi, PrintDpi,
            PixelFormats.Bgra32, null, raw.BgraPixels, raw.Stride);
        bitmap.Freeze();

        var container = new Grid
        {
            Width = _pageSize.Width,
            Height = _pageSize.Height,
            Background = Brushes.White
        };

        var image = new Image
        {
            Source = bitmap,
            Stretch = Stretch.Uniform,
            HorizontalAlignment = HorizontalAlignment.Center,
            VerticalAlignment = VerticalAlignment.Center
        };
        RenderOptions.SetBitmapScalingMode(image, BitmapScalingMode.HighQuality);
        container.Children.Add(image);

        var rect = new Rect(new Point(0, 0), _pageSize);
        container.Measure(_pageSize);
        container.Arrange(rect);
        container.UpdateLayout();

        return new DocumentPage(container, _pageSize, rect, rect);
    }
}

        if (Math.Abs(dx) < 0.01 && Math.Abs(dy) < 0.01) return;

        X = newX;
        Y = newY;
        Model.Bounds = new PdfRect(newX, newY, Width, Height);

        if (Model.InkPoints.Count > 0)

    }

    public static List<PdfRect> MergeCharRectsByLine(IReadOnlyList<PageCharInfo> chars, int startIdx, int endIdx)
    {
        var merged = new List<PdfRect>();
        int from = Math.Max(0, Math.Min(startIdx, endIdx));
        int to = Math.Min(chars.Count - 1, Math.Max(startIdx, endIdx));

        double? curX = null, curY = null, curRight = null, curBottom = null, curBaseline = null;

        for (int i = from; i <= to; i++)
        {
            var ch = chars[i];
            if (ch.Character == '\n' || ch.Bounds.Width <= 0)
            {
                FlushCurrent();
                continue;
            }

            bool sameLine = curBaseline.HasValue
                && Math.Abs(ch.OriginY - curBaseline.Value) <= Math.Max(3.5, ch.FontSize * 0.35)
                && ch.Bounds.X >= curRight!.Value - ch.FontSize
                && ch.Bounds.X <= curRight.Value + ch.FontSize * 2.5;

            if (!sameLine)
            {
                FlushCurrent();
                curX = ch.Bounds.X;
                curY = ch.Bounds.Y;
                curRight = ch.Bounds.X + ch.Bounds.Width;
                curBottom = ch.Bounds.Y + ch.Bounds.Height;
                curBaseline = ch.OriginY;
            }
            else
            {
                curX = Math.Min(curX!.Value, ch.Bounds.X);
                curY = Math.Min(curY!.Value, ch.Bounds.Y);
                curRight = Math.Max(curRight!.Value, ch.Bounds.X + ch.Bounds.Width);
                curBottom = Math.Max(curBottom!.Value, ch.Bounds.Y + ch.Bounds.Height);
            }
        }

        FlushCurrent();
        return merged;

        void FlushCurrent()
        {
            if (curX.HasValue)
            {
                merged.Add(new PdfRect(curX.Value, curY!.Value, curRight!.Value - curX.Value, curBottom!.Value - curY.Value));
                curX = curY = curRight = curBottom = curBaseline = null;
            }
        }
    }

        var found = await Task.Run(() =>
        {
            var list = new List<SearchMatchItem>();
            for (int p = 0; p < Pages.Count; p++)
            {
                if (token.IsCancellationRequested || _disposed) return list;

                var chars = _pageCharsCache.GetOrAdd(p, idx => _readerEngine.GetPageChars(idx));
                if (chars.Count == 0) continue;

                var sb = new StringBuilder(chars.Count);
                for (int i = 0; i < chars.Count; i++)
                    sb.Append(chars[i].Character == '\n' ? ' ' : chars[i].Character);

                string pageText = sb.ToString();
                int pos = 0;
                while ((pos = pageText.IndexOf(trimmed, pos, StringComparison.OrdinalIgnoreCase)) >= 0)
                {
                    var rects = MergeCharRectsByLine(chars, pos, pos + trimmed.Length - 1);

                    pos += trimmed.Length;
                }
            }
            return list;
        }, token);

        if (token.IsCancellationRequested || _disposed) return;

        _searchMatches.AddRange(found);
        if (_searchMatches.Count == 0)

    public async Task RequestVisiblePagesAsync(int firstVisible, int lastVisible, bool forceRerender = false)
    {
        if (_disposed || Pages.Count == 0) return;

        firstVisible = Math.Clamp(firstVisible, 0, Pages.Count - 1);
        lastVisible = Math.Clamp(lastVisible, firstVisible, Pages.Count - 1);

        _lastFirstVisible = firstVisible;
        _lastLastVisible = lastVisible;

        int start = Math.Max(0, firstVisible - 1);
        int end = Math.Min(Pages.Count - 1, lastVisible + 1);

        _renderCts?.Cancel();
        _renderCts = new CancellationTokenSource();
        var token = _renderCts.Token;

        double targetScale = PdfPageViewModel.DipPerPoint * (ZoomPercent / 100.0) * _currentDpiScale;

        for (int i = start; i <= end; i++)
        {
            if (token.IsCancellationRequested || _disposed) return;

            var pageVm = Pages[i];
            TouchLru(i);

            bool needsRender = forceRerender
                || pageVm.PageImage is null
                || Math.Abs(pageVm.RenderedScale - targetScale) > 0.05;

            if (!needsRender || !_renderingPages.Add(i))
                continue;

            int pageIndex = i;
            try
            {
                var bitmap = await Task.Run(() =>
                {
                    if (token.IsCancellationRequested || _disposed) return null;
                    RenderedPageResult raw = _readerEngine.RenderPage(pageIndex, targetScale);
                    return CreateFrozenBitmap(raw);
                }, token);

                if (bitmap is not null && !token.IsCancellationRequested && !_disposed)
                {
                    pageVm.PageImage = bitmap;
                    pageVm.RenderedScale = targetScale;
                    EvictOverflowPages(start, end);
                }
            }
            catch (OperationCanceledException) { return; }
            catch (ObjectDisposedException) { return; }
            finally { _renderingPages.Remove(pageIndex); }
        }
    }

    public async Task RequestVisibleThumbnailsAsync(int firstIndex, int lastIndex)
    {
        if (_disposed || Pages.Count == 0) return;

        int start = Math.Clamp(firstIndex, 0, Pages.Count - 1);
        int end = Math.Clamp(lastIndex, start, Pages.Count - 1);

        for (int i = start; i <= end; i++)
        {
            if (_disposed) return;

            var pageVm = Pages[i];
            if (pageVm.ThumbnailImage is not null || !_renderingThumbnails.Add(i))
                continue;

            int pageIndex = i;
            try
            {
                var thumb = await Task.Run(() =>
                {
                    if (_disposed) return null;
                    RenderedPageResult raw = _readerEngine.RenderThumbnail(pageIndex, 130);
                    return CreateFrozenBitmap(raw);
                });

                if (thumb is not null && !_disposed)
                    pageVm.ThumbnailImage = thumb;
            }
            catch (ObjectDisposedException) { return; }
            finally { _renderingThumbnails.Remove(pageIndex); }
        }
    }

    private void TouchLru(int pageIndex)
    {
        if (_lruSet.Contains(pageIndex)) _lruOrder.Remove(pageIndex);
        else _lruSet.Add(pageIndex);
        _lruOrder.AddFirst(pageIndex);
    }

    private void EvictOverflowPages(int keepStart, int keepEnd)
    {
        while (_lruOrder.Count > MaxCachedPages)
        {
            int oldest = _lruOrder.Last!.Value;
            if (oldest >= keepStart && oldest <= keepEnd) break;

            _lruOrder.RemoveLast();
            _lruSet.Remove(oldest);
            Pages[oldest].PageImage = null;
            Pages[oldest].RenderedScale = 0;
        }
    }

    private static BitmapSource CreateFrozenBitmap(RenderedPageResult raw)
    {
        var bmp = BitmapSource.Create(
            raw.Width, raw.Height, 96, 96,
            PixelFormats.Bgra32, null, raw.BgraPixels, raw.Stride);
        bmp.Freeze();
        return bmp;
    }

        double oldSize = target.FontSize;

                target.FontName = oldFont;
                target.FontSize = oldSize;
            }));
    }

    public static double ParseFontSize(string sizeLabel, double fallback = 13.0)
    {
        string digits = sizeLabel.Replace("pt", "", StringComparison.OrdinalIgnoreCase).Trim();
        return double.TryParse(digits, out double val) ? val : fallback;
    }

    private static List<int> ParsePageRanges(string input, int maxPage)
    {
        var result = new List<int>();
        foreach (string rawPart in input.Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries))
        {
            if (rawPart.Contains('-'))
            {
                string[] bounds = rawPart.Split('-', StringSplitOptions.TrimEntries);
                if (bounds.Length == 2
                    && int.TryParse(bounds[0], out int start)
                    && int.TryParse(bounds[1], out int end))
                {
                    int from = Math.Clamp(Math.Min(start, end), 1, maxPage);
                    int to = Math.Clamp(Math.Max(start, end), 1, maxPage);
                    for (int p = from; p <= to; p++) result.Add(p);
                }
            }
            else if (int.TryParse(rawPart, out int single) && single >= 1 && single <= maxPage)
            {
                result.Add(single);
            }
        }
        return result;
    }

    public void OpenDocumentFromPath(string filePath)
    {
        if (!File.Exists(filePath)) return;

        var existing = Tabs.FirstOrDefault(t =>
            string.Equals(t.FilePath, filePath, StringComparison.OrdinalIgnoreCase));

        if (existing is not null)
        {
            ActiveTab = existing;
            return;
        }

        try
        {
            var newTab = new DocumentTabViewModel(filePath);
            Tabs.Add(newTab);
            ActiveTab = newTab;
        }
        catch (Exception ex)
        {
            MessageBox.Show(
                $"KhÃ´ng thá»ƒ má»Ÿ tá»‡p \"{Path.GetFileName(filePath)}\".\nChi tiáº¿t: {ex.Message}",
                "Lá»—i má»Ÿ tá»‡p PDF",
                MessageBoxButton.OK,
                MessageBoxImage.Error);
        }
    }

        Tabs.Remove(tab);

        if (ActiveTab == tab)
        {
            ActiveTab = Tabs.Count > 0
                ? Tabs[Math.Clamp(index - 1, 0, Tabs.Count - 1)]
                : null;
        }
    }

namespace PdfStudio.Core;

public interface IUndoableAction
{
    void Execute();
    void Undo();
}

public sealed class UndoRedoManager
{
    private readonly Stack<IUndoableAction> _undoStack = new();
    private readonly Stack<IUndoableAction> _redoStack = new();

    public event Action? StateChanged;

    public bool CanUndo => _undoStack.Count > 0;
    public bool CanRedo => _redoStack.Count > 0;

    public void Execute(IUndoableAction action)
    {
        action.Execute();
        _undoStack.Push(action);
        _redoStack.Clear();
        StateChanged?.Invoke();
    }

    public void Undo()
    {
        if (_undoStack.Count == 0) return;
        var action = _undoStack.Pop();
        action.Undo();
        _redoStack.Push(action);
        StateChanged?.Invoke();
    }

    public void Redo()
    {
        if (_redoStack.Count == 0) return;
        var action = _redoStack.Pop();
        action.Execute();
        _undoStack.Push(action);
        StateChanged?.Invoke();
    }

    public void Clear()
    {
        _undoStack.Clear();
        _redoStack.Clear();
        StateChanged?.Invoke();
    }
}

    public static void Save(string inputPath, string outputPath, IReadOnlyList<PdfAnnotationModel> annotations)
    {
        if (annotations.Count == 0)
        {
            File.Copy(inputPath, outputPath, overwrite: true);
            return;
        }

        var reader = new PdfReader(inputPath);
        reader.SetUnethicalReading(true);

        using var writer = new PdfWriter(outputPath);
        // Incremental Update (AppendMode) giÃºp Ctrl+S cá»±c nhanh vÃ  giá»¯ nguyÃªn cáº¥u trÃºc gá»‘c
        using var pdfDoc = new PdfDocument(reader, writer, new StampingProperties().UseAppendMode());

        foreach (var item in annotations)
        {
            int pageNum = item.PageIndex + 1;
            if (pageNum < 1 || pageNum > pdfDoc.GetNumberOfPages()) continue;

            PdfPage page = pdfDoc.GetPage(pageNum);
            PdfAnnotation? annot = item.Type switch
            {
                AnnotationType.Highlight => CreateTextMarkupAnnotation(pdfDoc, page, item, isUnderline: false),
                AnnotationType.Underline => CreateTextMarkupAnnotation(pdfDoc, page, item, isUnderline: true),
                AnnotationType.StickyNote => CreateStickyNoteAnnotation(pdfDoc, page, item),
                AnnotationType.FreeText => CreateFreeTextAnnotation(pdfDoc, page, item),
                AnnotationType.Ink => CreateInkAnnotation(pdfDoc, page, item),
                _ => null
            };

            if (annot is not null)
                page.AddAnnotation(annot);
        }

    private static PdfAnnotation? CreateTextMarkupAnnotation(
        PdfDocument pdfDoc, PdfPage page, PdfAnnotationModel item, bool isUnderline)
    {
        var sourceRects = item.QuadRects.Count > 0 ? item.QuadRects : [item.Bounds];
        if (sourceRects.Count == 0) return null;

        var (r, g, b) = ParseHexColor(item.ColorHex, isUnderline ? (0.09f, 0.64f, 0.29f) : (0.98f, 0.80f, 0.08f));
        var rgbColor = new DeviceRgb(r, g, b);

        var pdfRects = new List<Rectangle>(sourceRects.Count);
        var quadValues = new List<float>(sourceRects.Count * 8);

        foreach (var vr in sourceRects)
        {
            var pr = ToPdfRect(page, vr);
            pdfRects.Add(pr);

            // Thá»© tá»± 4 Ä‘á»‰nh QuadPoints theo PDF Spec: TopLeft, TopRight, BottomLeft, BottomRight
            var tl = ToPdfPoint(page, vr.X, vr.Y);
            var tr = ToPdfPoint(page, vr.X + vr.Width, vr.Y);
            var bl = ToPdfPoint(page, vr.X, vr.Y + vr.Height);
            var br = ToPdfPoint(page, vr.X + vr.Width, vr.Y + vr.Height);

            quadValues.AddRange([tl.X, tl.Y, tr.X, tr.Y, bl.X, bl.Y, br.X, br.Y]);
        }

        Rectangle unionRect = UnionRects(pdfRects);
        var dict = new PdfDictionary();
        dict.Put(PdfName.Type, PdfName.Annot);
        dict.Put(PdfName.Subtype, isUnderline ? PdfName.Underline : PdfName.Highlight);
        dict.Put(PdfName.Rect, ToPdfArray(unionRect));
        dict.Put(new PdfName("QuadPoints"), new PdfArray(quadValues.ToArray()));
        dict.Put(PdfName.C, new PdfArray(new float[] { r, g, b }));
        dict.Put(new PdfName("CA"), new PdfNumber(isUnderline ? 1.0f : (float)Math.Clamp(item.Opacity, 0.15, 0.85)));
        dict.Put(PdfName.F, new PdfNumber(4));

        // Sinh Normal Appearance Stream Ä‘á»ƒ MuPDF/Chrome hiá»ƒn thá»‹ chuáº©n mÃ u & Ä‘á»™ trong suá»‘t
        var appearance = new PdfFormXObject(new Rectangle(0, 0, unionRect.GetWidth(), unionRect.GetHeight()));
        var canvas = new PdfCanvas(appearance, pdfDoc);
        canvas.SaveState();

        if (isUnderline)
        {
            canvas.SetStrokeColor(rgbColor);
            canvas.SetLineWidth((float)Math.Clamp(item.StrokeThickness, 1.0, 4.0));
            foreach (var pr in pdfRects)
            {
                float lx = pr.GetLeft() - unionRect.GetLeft();
                float ly = pr.GetBottom() - unionRect.GetBottom() + 1.0f;
                canvas.MoveTo(lx, ly).LineTo(lx + pr.GetWidth(), ly).Stroke();
            }
        }
        else
        {
            var gState = new PdfExtGState()
                .SetFillOpacity((float)Math.Clamp(item.Opacity, 0.15, 0.85))
                .SetBlendMode(PdfExtGState.BM_MULTIPLY);
            canvas.SetExtGState(gState);
            canvas.SetFillColor(rgbColor);
            foreach (var pr in pdfRects)
            {
                float lx = pr.GetLeft() - unionRect.GetLeft();
                float ly = pr.GetBottom() - unionRect.GetBottom();
                canvas.Rectangle(lx, ly, pr.GetWidth(), pr.GetHeight()).Fill();
            }
        }

        canvas.RestoreState();
        var annot = PdfAnnotation.MakeAnnotation(dict);
        annot.SetNormalAppearance(appearance.GetPdfObject());
        return annot;
    }

    private static PdfAnnotation? CreateInkAnnotation(PdfDocument pdfDoc, PdfPage page, PdfAnnotationModel item)
    {
        if (item.InkPoints.Count < 2) return null;

        var (r, g, b) = ParseHexColor(item.ColorHex, (0.72f, 0.11f, 0.11f));
        float lineWidth = (float)Math.Clamp(item.StrokeThickness, 0.75, 14.0);

        float minX = pdfPoints.Min(p => p.X) - lineWidth;
        float maxX = pdfPoints.Max(p => p.X) + lineWidth;
        float minY = pdfPoints.Min(p => p.Y) - lineWidth;
        float maxY = pdfPoints.Max(p => p.Y) + lineWidth;

        var rect = new Rectangle(minX, minY, Math.Max(2, maxX - minX), Math.Max(2, maxY - minY));

        var strokeCoords = new PdfArray();
        foreach (var pt in pdfPoints)
        {
            strokeCoords.Add(new PdfNumber(pt.X));
            strokeCoords.Add(new PdfNumber(pt.Y));
        }

        var inkList = new PdfArray();
        inkList.Add(strokeCoords);

        var borderStyle = new PdfDictionary();
        borderStyle.Put(new PdfName("W"), new PdfNumber(lineWidth));

        var dict = new PdfDictionary();
        dict.Put(PdfName.Type, PdfName.Annot);
        dict.Put(PdfName.Subtype, PdfName.Ink);
        dict.Put(PdfName.Rect, ToPdfArray(rect));

        dict.Put(PdfName.C, new PdfArray(new float[] { r, g, b }));
        dict.Put(new PdfName("BS"), borderStyle);
        dict.Put(PdfName.F, new PdfNumber(4));

        var canvas = new PdfCanvas(appearance, pdfDoc);
        canvas.SaveState();
        canvas.SetStrokeColor(new DeviceRgb(r, g, b));
        canvas.SetLineWidth(lineWidth);
        canvas.SetLineCapStyle(PdfCanvasConstants.LineCapStyle.ROUND);
        canvas.SetLineJoinStyle(PdfCanvasConstants.LineJoinStyle.ROUND);

        canvas.MoveTo(pdfPoints[0].X - rect.GetLeft(), pdfPoints[0].Y - rect.GetBottom());
        for (int i = 1; i < pdfPoints.Count; i++)
            canvas.LineTo(pdfPoints[i].X - rect.GetLeft(), pdfPoints[i].Y - rect.GetBottom());
        canvas.Stroke();
        canvas.RestoreState();

        var annot = PdfAnnotation.MakeAnnotation(dict);
        annot.SetNormalAppearance(appearance.GetPdfObject());
        return annot;
    }

    private static (float X, float Y) ToPdfPoint(PdfPage page, double vx, double vy)
    {
        Rectangle crop = page.GetCropBox();
        float cx = crop.GetLeft();
        float cy = crop.GetBottom();
        float cw = crop.GetWidth();
        float ch = crop.GetHeight();
        int rot = ((page.GetRotation() % 360) + 360) % 360;

        return rot switch
        {
            90 => ((float)(cx + vy), (float)(cy + vx)),
            180 => ((float)(cx + cw - vx), (float)(cy + vy)),
            270 => ((float)(cx + cw - vy), (float)(cy + ch - vx)),
            _ => ((float)(cx + vx), (float)(cy + ch - vy))
        };
    }

    private static Rectangle ToPdfRect(PdfPage page, PdfRect vr)
    {
        var p1 = ToPdfPoint(page, vr.X, vr.Y);
        var p2 = ToPdfPoint(page, vr.X + vr.Width, vr.Y);
        var p3 = ToPdfPoint(page, vr.X, vr.Y + vr.Height);
        var p4 = ToPdfPoint(page, vr.X + vr.Width, vr.Y + vr.Height);

        float minX = Math.Min(Math.Min(p1.X, p2.X), Math.Min(p3.X, p4.X));
        float maxX = Math.Max(Math.Max(p1.X, p2.X), Math.Max(p3.X, p4.X));
        float minY = Math.Min(Math.Min(p1.Y, p2.Y), Math.Min(p3.Y, p4.Y));
        float maxY = Math.Max(Math.Max(p1.Y, p2.Y), Math.Max(p3.Y, p4.Y));

        return new Rectangle(minX, minY, Math.Max(1, maxX - minX), Math.Max(1, maxY - minY));

        float minY = rects.Min(r => r.GetBottom());
        float maxX = rects.Max(r => r.GetRight());
        float maxY = rects.Max(r => r.GetTop());
        return new Rectangle(minX, minY, Math.Max(1, maxX - minX), Math.Max(1, maxY - minY));

    private static (float R, float G, float B) ParseHexColor(string hex, (float, float, float) fallback)
    {
        if (string.IsNullOrWhiteSpace(hex)) return fallback;
        string clean = hex.Trim().TrimStart('#');
        if (clean.Length == 8) clean = clean[2..]; // Bá» kÃªnh alpha náº¿u cÃ³ dáº¡ng #AARRGGBB
        if (clean.Length != 6) return fallback;

        if (byte.TryParse(clean[0..2], NumberStyles.HexNumber, null, out byte r)
            && byte.TryParse(clean[2..4], NumberStyles.HexNumber, null, out byte g)
            && byte.TryParse(clean[4..6], NumberStyles.HexNumber, null, out byte b))
        {
            return (r / 255f, g / 255f, b / 255f);
        }
        return fallback;
    }

    private static PdfFont LoadVietnameseFont(string? fontName)
    {

            "calibri" => ["calibri.ttf", "arial.ttf"],
            "tahoma" => ["tahoma.ttf", "arial.ttf"],
            "segoe ui" => ["segoeui.ttf", "arial.ttf"],
            _ => ["arial.ttf", "segoeui.ttf", "tahoma.ttf"]
        };

        foreach (string file in candidates)
        {
            string fullPath = System.IO.Path.Combine(fontsDir, file);
            if (File.Exists(fullPath))
                return PdfFontFactory.CreateFont(fullPath, PdfEncodings.IDENTITY_H, PdfFontFactory.EmbeddingStrategy.PREFER_EMBEDDED);
        }

        return PdfFontFactory.CreateFont(StandardFonts.HELVETICA);

    {
        lock (_syncLock)
        {
            ThrowIfDisposed();
            if (pageIndex < 0 || pageIndex >= PageCount)
                throw new ArgumentOutOfRangeException(nameof(pageIndex));

            float zoom = (float)Math.Clamp(scaleFactor, 0.1, 6.0);
            Rectangle bounds = _document!.Pages[pageIndex].Bounds;

            byte[] pixels = _document.Render(

                true);

            int stride = width * 4;
            int height = Math.Max(1, Math.Min(rounded.Height, pixels.Length / stride));

            return new RenderedPageResult(width, height, stride, pixels);

                        double gap = x - (prev.Bounds.X + prev.Bounds.Width);

namespace PdfStudio.Engines.QPdf;

public sealed class QPdfCliService
{
    private readonly string _qpdfPath;

    public QPdfCliService()
    {
        _qpdfPath = Path.Combine(AppContext.BaseDirectory, "Tools", "qpdf.exe");
    }

    public bool IsAvailable => File.Exists(_qpdfPath);

    public bool TryCompress(string inputPath, string outputPath)
    {
        if (!IsAvailable) return false;

        var startInfo = new ProcessStartInfo
        {
            FileName = _qpdfPath,
            ArgumentList =
            {
                "--object-streams=generate",
                "--compress-streams=y",
                "--recompress-flate",
                "--compression-level=9",
                inputPath,
                outputPath
            },
            UseShellExecute = false,
            CreateNoWindow = true,
            RedirectStandardError = true
        };

        using var proc = Process.Start(startInfo);
        if (proc is null) return false;

        proc.WaitForExit(60_000);
        // qpdf tráº£ mÃ£ 0 (thÃ nh cÃ´ng) hoáº·c 3 (thÃ nh cÃ´ng kÃ¨m cáº£nh bÃ¡o nháº¹ vá» cáº¥u trÃºc gá»‘c)
        return (proc.ExitCode == 0 || proc.ExitCode == 3) && File.Exists(outputPath);
    }
